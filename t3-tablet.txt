import { useRef, useState } from 'react';
import { useRouter } from 'next/navigation';
import SignatureCanvas from 'react-signature-canvas';
import { fmt, type Operation } from '@/lib/laravel-types';

const input =
  'h-14 w-full rounded-lg border border-gray-300 bg-white px-4 text-base focus:border-accent focus:outline-none focus:ring-2 focus:ring-red-200';
const card = 'space-y-4 rounded-xl border border-[#e9ecef] bg-white p-5';

function Field({ label, error, children }: { label: string; error?: string; children: React.ReactNode }) {
  return (
    <label className="block">
      <span className="mb-1 block text-xs font-semibold uppercase tracking-wide text-gray-500">{label}</span>
      {children}
      {error && <span className="mt-1 block text-sm text-red-600">{error}</span>}
    </label>
  );
}

export default function NovaDescargaForm({ operation }: { operation: Operation }) {
  const router = useRouter();
  const sig = useRef<SignatureCanvas>(null);
  const [qtys, setQtys] = useState<Record<number, string>>({});
  const [errors, setErrors] = useState<Record<string, string[]>>({});
  const [busy, setBusy] = useState(false);

  const err = (k: string) => errors[k]?.[0];
  const hoje = new Date().toLocaleDateString('en-CA');

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (busy) return;

    if (!sig.current || sig.current.isEmpty()) {
      setErrors({ assinatura: ['A assinatura do motorista é obrigatória.'] });
      return;
    }

    const f = new FormData(e.currentTarget);
    setBusy(true);
    setErrors({});

    try {
      const res = await fetch('/api/laravel/discharges', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          operation_id: operation.id,
          data: f.get('data'),
          motorista: f.get('motorista'),
          matricula: f.get('matricula'),
          transportadora: f.get('transportadora'),
          numero_guia: f.get('numero_guia'),
          tipo_atrelado: f.get('tipo_atrelado'),
          items: qtys,
          assinatura: sig.current.getCanvas().toDataURL('image/png'),
        }),
      });

      if (res.status === 401) return router.push('/login');
      if (res.status === 422) {
        setErrors((await res.json()).errors ?? {});
        return;
      }
      if (!res.ok) {
        setErrors({ geral: ['Falha ao registar. Verifique a ligação e tente novamente.'] });
        return;
      }

      const msg = encodeURIComponent('Camião registado como carregado. Aguarda a saída do porto.');
      router.push(`/operacoes/${operation.id}?ok=${msg}`);
      router.refresh();
    } catch {
      setErrors({ geral: ['Sem ligação ao servidor. Verifique a rede e tente novamente.'] });
    } finally {
      setBusy(false);
    }
  }

  return (
    <form onSubmit={onSubmit} className="mx-auto max-w-3xl space-y-5 p-6">
      {err('geral') && <p className="rounded-lg bg-red-50 p-3 text-red-700">{err('geral')}</p>}

      <section className={card}>
        <h2 className="font-display text-sm font-bold uppercase tracking-wide text-gray-500">Carregamento</h2>
        <Field label="Data">
          <input type="date" name="data" defaultValue={hoje} required className={input} suppressHydrationWarning />
        </Field>
        <p className="text-sm text-gray-500">A hora de carregamento fica registada automaticamente pelo servidor.</p>
      </section>

      <section className={card}>
        <h2 className="font-display text-sm font-bold uppercase tracking-wide text-gray-500">Camião</h2>
        <Field label="Motorista" error={err('motorista')}>
          <input name="motorista" required className={input} />
        </Field>
        <div className="grid grid-cols-2 gap-4">
          <Field label="Matrícula" error={err('matricula')}>
            <input name="matricula" required className={input} />
          </Field>
          <Field label="Número da guia" error={err('numero_guia')}>
            <input name="numero_guia" required className={input} />
          </Field>
        </div>
        <Field label="Transportadora" error={err('transportadora')}>
          <input name="transportadora" list="transportadoras" required className={input} />
        </Field>
        <datalist id="transportadoras">
          {operation.transportadoras.map((t) => (
            <option key={t} value={t} />
          ))}
        </datalist>
        <Field label="Tipo de atrelado" error={err('tipo_atrelado')}>
          <select name="tipo_atrelado" defaultValue="normal" className={input}>
            <option value="normal">Atrelado normal</option>
            <option value="link">Atrelado link</option>
          </select>
        </Field>
      </section>

      <section className={card}>
        <h2 className="font-display text-sm font-bold uppercase tracking-wide text-gray-500">Quantidades</h2>
        {operation.items.map((it) => (
          <div key={it.id}>
            <label className="flex items-center justify-between gap-4">
              <span className="text-slate-800">
                <span className="font-semibold">{it.nome}</span>
                <span className="block text-xs text-gray-500">
                  Saldo: {fmt(it.saldo)} {it.unidade_label}
                </span>
              </span>
              <input
                inputMode="decimal"
                value={qtys[it.id] ?? ''}
                onChange={(e) => setQtys({ ...qtys, [it.id]: e.target.value })}
                className={`${input} !w-40 text-right`}
              />
            </label>
            {err(`items.${it.id}`) && <p className="mt-1 text-sm text-red-600">{err(`items.${it.id}`)}</p>}
          </div>
        ))}
        {err('items') && <p className="text-sm text-red-600">{err('items')}</p>}
      </section>

      <section className={card}>
        <h2 className="font-display text-sm font-bold uppercase tracking-wide text-gray-500">Assinatura do motorista</h2>
        <div className="overflow-x-auto">
          <SignatureCanvas
            ref={sig}
            canvasProps={{
              width: 640,
              height: 200,
              style: { width: 640, height: 200, touchAction: 'none' },
              className: 'rounded-lg border border-gray-300 bg-white',
            }}
          />
        </div>
        <button type="button" onClick={() => sig.current?.clear()} className="h-11 rounded-lg border border-gray-300 px-4 text-sm font-medium text-gray-700">
          Limpar assinatura
        </button>
        {err('assinatura') && <p className="text-sm text-red-600">{err('assinatura')}</p>}
      </section>

      <button disabled={busy} className="h-16 w-full rounded-xl bg-accent text-lg font-semibold text-white shadow-sm disabled:opacity-50">
        {busy ? 'A registar...' : 'Registar camião'}
      </button>
    </form>
  );
}

function paginas(atual: number, ultima: number): (number | '...')[] {
  const out: (number | '...')[] = [];
  const ini = Math.max(1, atual - 1);
  const fim = Math.min(ultima, atual + 1);
  if (ini > 1) out.push(1);
  if (ini > 2) out.push('...');
  for (let i = ini; i <= fim; i++) out.push(i);
  if (fim < ultima - 1) out.push('...');
  if (fim < ultima) out.push(ultima);
  return out;
}

function paginas(atual: number, ultima: number): (number | '...')[] {
  const out: (number | '...')[] = [];
  const ini = Math.max(1, atual - 1);
  const fim = Math.min(ultima, atual + 1);
  if (ini > 1) out.push(1);
  if (ini > 2) out.push('...');
  for (let n = ini; n <= fim; n++) out.push(n);
  if (fim < ultima - 1) out.push('...');
  if (fim < ultima) out.push(ultima);
  return out;

export async function DELETE(_: Request, { params }: Ctx) {
  const id = Number((await params).id);
  return mutate((db) => {
    if (!db.operations.some((o) => o.id === id)) return notFound();
    if (db.discharges.some((d) => d.operationId === id)) return invalid(['Só pode eliminar operações sem descargas. Arquive-a em vez disso.']);
    db.operations = db.operations.filter((o) => o.id !== id);
    return ok({ ok: true });
  });

const LIMIAR_REL = 0.35;

export default function LinhaCamiao({

  const [erro, setErro] = useState<string | null>(null);
  const [dx, setDx] = useState(0);
  const [arrastando, setArrastando] = useState(false);
  const [passou, setPassou] = useState(false);
  const dxRef = useRef(0);
  const inicio = useRef<{ x: number; y: number; id: number; largura: number; ativo: boolean } | null>(null);

  const ocupado = enviando || pendente;
  const carregado = d.status === 'loaded';
  const podeDeslizar = carregado && !ocupado;

  // Quando o estado muda (depois de actualizar a lista), a linha volta à posição normal.
  useEffect(() => {
    dxRef.current = 0;
    setDx(0);
    setPassou(false);
    setErro(null);
  }, [d.status]);

  function repor() {
    dxRef.current = 0;
    setDx(0);

        return;
      }
      if (!res.ok) {
        const j = await res.json().catch(() => ({}));
        setErro(j?.errors?.status?.[0] ?? j?.message ?? `Erro ${res.status} ao registar a etapa.`);
        repor();
        return;
      }
      iniciar(() => router.refresh());
    } catch {
      setErro('Sem ligação ao servidor. Verifique a rede e tente novamente.');
      repor();
    } finally {
      setEnviando(false);
    }

    if (!podeDeslizar) return;
    if (e.pointerType === 'mouse' && e.button !== 0) return;
    inicio.current = { x: e.clientX, y: e.clientY, id: e.pointerId, largura: e.currentTarget.offsetWidth, ativo: false };
  }

  function aoMover(e: ReactPointerEvent<HTMLDivElement>) {
    const i = inicio.current;
    if (!i || e.pointerId !== i.id) return;
    const mx = e.clientX - i.x;
    const my = e.clientY - i.y;

    if (!i.ativo) {
      if (Math.abs(mx) < 10 && Math.abs(my) < 10) return;

export type Sla = '' | 'ok' | 'warn' | 'danger';

export function desdeSaida(d: Discharge): Date | null {
  if (!['pending', 'in_transit'].includes(d.status) || !d.horaSaidaPorto) return null;
  return new Date(`${d.data}T${d.horaSaidaPorto.slice(0, 5)}:00`);
}
export function slaMinutos(d: Discharge, now: number): number | null {
  const desde = desdeSaida(d);
  return desde ? Math.max(0, Math.floor((now - desde.getTime()) / 60000)) : null;
}
export function slaNivel(min: number | null): Sla {
  if (min === null) return '';
  return min >= SLA_VERMELHO ? 'danger' : min >= SLA_AMARELO ? 'warn' : 'ok';
}

/* ── Balança ── */
export function labelBalanca(d: Discharge) {
  if (d.status === 'weighing') return d.pesoBruto !== null ? 'Pesar vazio' : 'Pesar cheio';
  return 'Balança';
}
export function tempoTransporte(saida: string | null, chegada: string) {
  if (!saida || !chegada) return null;
  const [hd, md] = saida.split(':').map(Number);
  const [ha, ma] = chegada.split(':').map(Number);
  const diff = ha * 60 + ma - (hd * 60 + md);
  return diff >= 0 ? diff : null;
}
export function passosPesagem(bruto: string, tara: string) {
  const b = bruto.trim() !== '', t = tara.trim() !== '';
  const n = (b ? 1 : 0) + (t ? 1 : 0);
  const texto =
    n === 0 ? 'Registe pelo menos um peso (cheio ou vazio)'
    : n === 2 ? 'Pesagem completa: ao guardar, a descarga fica confirmada'
    : b ? 'Peso cheio registado. Falta pesar o camião vazio (ao voltar)'
    : 'Peso vazio registado. Falta pesar o camião cheio';
  return { b, t, n, texto };
}
export function validarPesos(bruto: string, tara: string): string | null {
  const sb = bruto.trim(), st = tara.trim();
  if (!sb && !st) return 'Registe pelo menos um peso (cheio ou vazio).';
  if (sb && st && parseFloat(sb) <= parseFloat(st)) return 'O peso cheio tem de ser maior que o peso vazio.';
  return null;
}

export interface Filters { de: string; ate: string; tr: string; st: '' | DischargeStatus; tc: string; dv: '' | 'sim' | 'nao'; at: '' | 'normal' | 'link' | 'nd' }
export const EMPTY_FILTERS: Filters = { de: '', ate: '', tr: '', st: '', tc: '', dv: '', at: '' };

export function applyFilters(list: Discharge[], f: Filters) {
  return list.filter((d) => {
    const prods = d.items.map((i) => i.nome.toLowerCase());
    if (f.de && d.data < f.de) return false;
    if (f.ate && d.data > f.ate) return false;
    if (f.tr && !d.transportadora.toLowerCase().includes(f.tr.toLowerCase())) return false;
    if (f.st && d.status !== f.st) return false;
    if (f.tc === 'misto' && prods.length < 2) return false;
    if (f.tc && f.tc !== 'misto' && !(prods.length === 1 && prods[0] === f.tc)) return false;
    if (f.dv && (d.hasDivergence ? 'sim' : 'nao') !== f.dv) return false;
    if (f.at && (d.tipoAtrelado ?? 'nd') !== f.at) return false;
    return true;
  });

const r3 = (n: number) => Math.round(n * 1000) / 1000;
const LABEL = { pending: 'Pendente', in_transit: 'Em trânsito', weighing: 'A pesar', confirmed: 'Confirmado' } as const;
const pad = (n: number) => String(n).padStart(2, '0');

export function diasNoParque(c: ContentorRow): number | null {
  if (!c.dataEntrada) return null;
  const ini = new Date(c.dataEntrada.slice(0, 10) + 'T00:00:00').getTime();
  const fim = c.dataSaida ? new Date(c.dataSaida.slice(0, 10) + 'T00:00:00').getTime() : Date.now();
  return Math.max(0, Math.floor((fim - ini) / 86400000));
}
export const toContentor = (c: ContentorRow): Contentor => ({ ...c, diasNoParque: diasNoParque(c) });

export function contentoresData(rows: ContentorRow[]): ContentoresData {
  const list = rows.map(toContentor);
  const ativo = (c: Contentor) => !['em_transito', 'saiu'].includes(c.estado);
  const noParque = list.filter(ativo);

  const meses: string[] = [], entradasMensais: number[] = [];
  const hoje = new Date();
  for (let i = 11; i >= 0; i--) {
    const d = new Date(hoje.getFullYear(), hoje.getMonth() - i, 1);
    meses.push(d.toLocaleDateString('pt-PT', { month: 'short', year: '2-digit' }).replace('.', ''));
    const key = `${d.getFullYear()}-${pad(d.getMonth() + 1)}`;
    entradasMensais.push(list.filter((c) => c.dataEntrada?.startsWith(key)).length);
  }

  const porEstadoMap: Record<string, number> = {};
  list.forEach((c) => { const n = ESTADOS[c.estado] ?? c.estado; porEstadoMap[n] = (porEstadoMap[n] ?? 0) + 1; });
  const topMap: Record<string, number> = {};
  noParque.forEach((c) => { if (c.cliente) topMap[c.cliente] = (topMap[c.cliente] ?? 0) + 1; });

  return {
    contentores: list,
    kpis: {
      total: list.length, noParque: noParque.length,
      emTransito: list.filter((c) => c.estado === 'em_transito').length,
      carregados: list.filter((c) => c.estado === 'carregado').length,
      saiu: list.filter((c) => c.estado === 'saiu').length,
      atrasados: noParque.filter((c) => (c.diasNoParque ?? 0) > LIMITE_DIAS).length,
      limite: LIMITE_DIAS,
      peso: noParque.reduce((s, c) => s + (c.pesoKg ?? 0), 0) / 1000,
    },
    meses, entradasMensais,
    porEstado: Object.entries(porEstadoMap).map(([nome, total]) => ({ nome, total })),
    topClientes: Object.entries(topMap).map(([nome, total]) => ({ nome, total })).sort((a, b) => b.total - a.total).slice(0, 8),
    estados: ESTADOS, tipos: TIPOS,
  };
}

export function toDischarge(d: DischargeRow, op: OperationRow): Discharge {
  let estimado = 0, sacos = 0, sacosConf = 0;
  const items = d.items.map((i) => {
    const p = op.planItems.find((x) => x.id === i.operationItemId);
    if (p?.unidade === 'saco') { estimado += i.quantidade * (p.pesoUnitarioKg ?? 0); sacos += i.quantidade; sacosConf += i.quantidadeConfirmada ?? i.quantidade; }
    else if (p) estimado += i.quantidade * 1000;
    return { operationItemId: i.operationItemId, nome: p?.nome ?? '?', unidade: p?.unidade ?? ('saco' as const), quantidade: i.quantidade, quantidadeConfirmada: i.quantidadeConfirmada };
  });
  const liquido = d.pesoBruto !== null && d.tara !== null ? d.pesoBruto - d.tara : null;
  let tempo: number | null = null;
  if (d.horaSaidaPorto && d.horaChegadaBalanca) {
    const [h1, m1] = d.horaSaidaPorto.split(':').map(Number), [h2, m2] = d.horaChegadaBalanca.split(':').map(Number);
    const diff = h2 * 60 + m2 - (h1 * 60 + m1);
    tempo = diff >= 0 ? diff : null;
  }
  const acima = estimado / 1000 > LIMITE_ATRELADO_T;
  return {
    id: d.id, data: d.data, numeroGuia: d.numeroGuia, motorista: d.motorista, matricula: d.matricula,
    transportadora: d.transportadora, assinado: d.assinado, tipoAtrelado: d.tipoAtrelado,
    atreladoInconsistente: d.tipoAtrelado ? (acima ? d.tipoAtrelado === 'normal' : d.tipoAtrelado === 'link') : false,
    status: d.status, statusLabel: LABEL[d.status],
    horaSaidaPorto: d.horaSaidaPorto, horaChegadaBalanca: d.horaChegadaBalanca,
    pesoBruto: d.pesoBruto, tara: d.tara, pesoLiquido: liquido !== null && liquido > 0 ? liquido : null,
    pesoTotalEstimado: estimado, tempoTransporte: tempo, totalSacos: sacos, sacosConfirmados: sacosConf,
    hasDivergence: items.some((i) => i.quantidadeConfirmada !== null && Math.abs(i.quantidadeConfirmada - i.quantidade) > 0.0005),
    observacoes: d.observacoes, items,
  };
}

export function toOperation(op: OperationRow, ds: DischargeRow[]): Operation {
  const esperada = op.planItems.reduce((s, p) => s + (p.unidade === 'saco' ? (p.quantidade * (p.pesoUnitarioKg ?? 0)) / 1000 : p.quantidade), 0);
  const recebido = ds.filter((d) => d.status === 'confirmed' && d.pesoBruto !== null && d.tara !== null)
    .reduce((s, d) => s + Math.max(0, (d.pesoBruto as number) - (d.tara as number)), 0) / 1000;
  return {
    id: op.id, navio: op.navio, empresa: op.empresa, bl: op.bl, factura: op.factura,
    dataChegada: op.dataChegada, startedAt: op.startedAt, archivedAt: op.archivedAt, archived: !!op.archivedAt,
    quantidadeEsperada: r3(esperada), recebidoTon: r3(recebido), faltaTon: r3(esperada - recebido),
    progresso: esperada > 0 ? Math.round((recebido / esperada) * 100) : 0,
    metadata: op.metadata, observacoes: op.observacoes, transportadoras: op.transportadoras,
    transportadorasExtra: op.transportadoras.filter((t) => t.extra).map((t) => t.nome.toLowerCase()),
    planItems: op.planItems.map((p) => ({
      id: p.id, nome: p.nome, quantidade: p.quantidade, unidade: p.unidade, pesoUnitarioKg: p.pesoUnitarioKg,
      registado: r3(ds.flatMap((d) => d.items).filter((i) => i.operationItemId === p.id).reduce((s, i) => s + i.quantidade, 0)),
      confirmado: r3(ds.filter((d) => d.status === 'confirmed').flatMap((d) => d.items).filter((i) => i.operationItemId === p.id).reduce((s, i) => s + (i.quantidadeConfirmada ?? 0), 0)),
    })),
  };
}

import type { ContentorRow, DB, DischargeRow, OperationRow } from './db';
import { ESTADOS, TIPOS } from './constants';
import { Rec, dt, hhmm, num, str } from './http';

export function validateContentor(b: Rec, db: DB, ignoreId?: number) {
  const errors: string[] = [];
  const numero = (str(b.numero) ?? '').toUpperCase().replace(/\s/g, '');
  if (!/^[A-Z]{4}\d{7}$/.test(numero)) errors.push('O número do contentor deve ter 4 letras e 7 dígitos (ISO 6346).');
  else if (db.contentores.some((c) => c.numero === numero && c.id !== ignoreId)) errors.push('Já existe um contentor com este número.');
  const tipo = str(b.tipo) ?? '';
  if (!TIPOS.includes(tipo)) errors.push('Tipo inválido.');
  const estado = str(b.estado) ?? '';
  if (!(estado in ESTADOS)) errors.push('Estado inválido.');
  const pesoKg = num(b.pesoKg);
  if (Number.isNaN(pesoKg) || (pesoKg !== null && pesoKg < 0)) errors.push('Peso inválido.');
  const dataEntrada = dt(b.dataEntrada), dataSaida = dt(b.dataSaida);
  if (dataEntrada === undefined || dataSaida === undefined) errors.push('Data inválida.');
  else if (dataEntrada && dataSaida && dataSaida < dataEntrada) errors.push('A data de saída não pode ser anterior à entrada.');
  const value: Omit<ContentorRow, 'id'> = {
    numero, tipo, estado, armador: str(b.armador), cliente: str(b.cliente), navio: str(b.navio), bl: str(b.bl), selo: str(b.selo),
    pesoKg: pesoKg === null || Number.isNaN(pesoKg) ? null : pesoKg, localizacao: str(b.localizacao),
    dataEntrada: dataEntrada ?? null, dataSaida: dataSaida ?? null, observacoes: str(b.observacoes),
  };
  return { errors, value };
}

export function validateOperation(b: Rec) {
  const errors: string[] = [];
  const navio = str(b.navio), empresa = str(b.empresa);
  if (!navio) errors.push('Indique o nome do navio.');
  if (!empresa) errors.push('Indique a empresa.');
  const dataChegada = dt(b.dataChegada);
  if (dataChegada === undefined) errors.push('Data de chegada inválida.');
  const transp = (Array.isArray(b.transportadoras) ? b.transportadoras : []).map(str).filter(Boolean) as string[];
  const items: { id: number | null; nome: string; quantidade: number; unidade: 'saco' | 'tonelada'; pesoUnitarioKg: number | null }[] = [];
  (Array.isArray(b.planItems) ? b.planItems : []).forEach((p: Rec, i: number) => {
    const nome = str(p.nome), q = num(p.quantidade), peso = num(p.pesoUnitarioKg);
    const unidade = p.unidade === 'tonelada' ? 'tonelada' : 'saco';
    if (!nome) return errors.push(`Produto ${i + 1}: indique o nome.`) as unknown as void;
    if (q === null || Number.isNaN(q) || q <= 0) return errors.push(`${nome}: quantidade inválida.`) as unknown as void;
    if (unidade === 'saco' && (peso === null || Number.isNaN(peso) || peso <= 0)) return errors.push(`${nome}: indique o peso do saco (kg).`) as unknown as void;
    items.push({ id: Number(p.id) || null, nome, quantidade: q, unidade, pesoUnitarioKg: unidade === 'saco' ? (peso as number) : null });
  });
  if (!items.length && !errors.length) errors.push('Adicione pelo menos um produto ao plano.');
  return {
    errors, transp, items,
    value: { navio: navio ?? '', empresa: empresa ?? '', bl: str(b.bl), factura: str(b.factura), dataChegada: dataChegada ?? null, observacoes: str(b.observacoes) },
  };
}

export function validateDischarge(b: Rec, op: OperationRow) {
  const errors: string[] = [];
  const data = dt(b.data);
  if (!data) errors.push('Indique a data da descarga.');
  const motorista = str(b.motorista), matricula = str(b.matricula), transportadora = str(b.transportadora);
  if (!motorista) errors.push('Indique o motorista.');
  if (!matricula) errors.push('Indique a matrícula.');
  if (!transportadora) errors.push('Indique a transportadora.');
  const saida = hhmm(b.horaSaidaPorto);
  if (saida === undefined) errors.push('Hora de saída inválida.');
  const items: { operationItemId: number; quantidade: number }[] = [];
  (Array.isArray(b.items) ? b.items : []).forEach((i: Rec) => {
    const id = Number(i.operationItemId), q = num(i.quantidade);
    const p = op.planItems.find((x) => x.id === id);
    if (!p || q === null || Number.isNaN(q) || q <= 0) return;
    if (p.unidade === 'saco' && !Number.isInteger(q)) errors.push(`${p.nome}: o nº de sacos tem de ser inteiro.`);
    items.push({ operationItemId: id, quantidade: q });
  });
  if (!items.length) errors.push('Indique a quantidade de pelo menos um produto.');
  const tipoAtrelado = b.tipoAtrelado === 'normal' || b.tipoAtrelado === 'link' ? (b.tipoAtrelado as 'normal' | 'link') : null;
  return {
    errors, items,
    value: {
      data: data ?? '', numeroGuia: str(b.numeroGuia), motorista: motorista ?? '', matricula: matricula ?? '',
      transportadora: transportadora ?? '', assinado: !!b.assinado, tipoAtrelado,
      horaSaidaPorto: saida ?? null, observacoes: str(b.observacoes),
    },
  };
}
